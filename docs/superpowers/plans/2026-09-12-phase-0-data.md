# Phase 0 Data Pipeline Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A repeatable pipeline that, given a WoW client build, downloads the DB2 tables we need from wago.tools, normalizes them into stable JSON the Astro site can read, records a manifest with checksums, and can diff two builds into changelog-ready entries. Proven on the current Classic Era build before Sept 17 so it can run against the Forever beta the day it appears.

**Architecture:** A small Python package, `pipeline`, with three commands: `fetch` (download CSV exports for a build into `data/builds/<build>/raw/`, gitignored), `normalize` (read raw CSVs, write `data/builds/<build>/<entity>.json` plus `manifest.json`, committed), and `diff` (compare two builds' normalized JSON and write `data/diffs/<from>__<to>.json`). Every normalizer is a pure function from rows to records with a golden-file test.

**Tech Stack:** Python 3.12, `uv` for env and deps, `httpx` for downloads, `pydantic` v2 for record models, `pytest`, `ruff`. No database. Runs locally and in GitHub Actions.

**Spec:** `docs/superpowers/specs/2026-09-12-forever-sixty-phase-0-design.md`

## Global Constraints

- wago.tools CSV export URL pattern: `https://wago.tools/db2/<Table>/csv?build=<build>` where `<build>` is like `1.15.7.61582`. Build lists: `https://wago.tools/api/builds` (JSON keyed by product, e.g. `wow_classic_era`). The Forever product key is unknown until Sept 17; it is a CLI argument, never hardcoded.
- Tables fetched in Phase 0: `Map`, `AreaTable`, `JournalInstance`, `ItemSparse`, `Item`, `SpellName`, `ChrClasses`, `ChrRaces`, `Talent`, `TalentTab`.
- Normalized JSON is deterministic: records sorted by `id`, keys in model order, `indent=2`, trailing newline. Two runs on the same raw input produce byte-identical files.
- Every normalized file carries no build info inside; `manifest.json` holds `build`, `product`, `fetched_at`, and `sha256` per file.
- Raw CSVs are never committed (`data/builds/*/raw/` is gitignored). Normalized JSON is committed while under ~50 MB per build; the plan notes the switch to R2 if it grows.
- Network access happens only in `fetch`. `normalize` and `diff` are offline and fully unit-tested.
- Coverage target 80% on `pipeline/`.
- Commits use conventional format and end with the two trailer lines:
  ```
  Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>
  Claude-Session: https://claude.ai/code/session_01EkkERdonhxS2PMgXD7qZ6k
  ```
- Run every command from `data/`.

```python
# data/pipeline/__main__.py
import argparse
import sys


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(prog="pipeline", description="Forever Sixty data pipeline")
    sub = p.add_subparsers(dest="command", required=True)

    f = sub.add_parser("fetch", help="download DB2 CSV exports for a build")
    f.add_argument("--product", required=True, help="wago.tools product key, e.g. wow_classic_era")
    f.add_argument("--build", help="build string like 1.15.7.61582; default: latest for product")

    n = sub.add_parser("normalize", help="normalize raw CSVs into JSON")
    n.add_argument("--build", required=True)

    d = sub.add_parser("diff", help="diff two normalized builds")
    d.add_argument("--from", dest="from_build", required=True)
    d.add_argument("--to", dest="to_build", required=True)
    return p


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    if args.command == "fetch":
        from pipeline.wago import fetch_build

        fetch_build(args.product, args.build)
    elif args.command == "normalize":
        from pipeline.normalize import normalize_build

        normalize_build(args.build)
    elif args.command == "diff":
        from pipeline.diff import diff_builds

        diff_builds(args.from_build, args.to_build)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# data/pipeline/normalize/__init__.py
import json
from collections.abc import Sequence
from pathlib import Path

from pydantic import BaseModel


def write_json(records: Sequence[BaseModel], path: Path) -> None:
    ordered = sorted(records, key=lambda r: r.id)  # type: ignore[attr-defined]
    payload = [r.model_dump() for r in ordered]
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(payload, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")


def normalize_build(build: str) -> None:  # completed in Task 6
    raise NotImplementedError(build)
```

```python
# data/pipeline/normalize/zones.py
from pipeline.models import Zone


def _int_or_none(v: str) -> int | None:
    v = v.strip()
    return int(v) if v not in ("", "0") else None


def normalize_zones(area_rows: list[dict[str, str]], map_rows: list[dict[str, str]]) -> list[Zone]:
    maps = {int(m["ID"]): m["MapName_lang"] for m in map_rows}
    zones: list[Zone] = []
    for a in area_rows:
        map_id = int(a["ContinentID"])
        level = int(a["ExplorationLevel"]) if a["ExplorationLevel"].strip() else 0
        zones.append(
            Zone(
                id=int(a["ID"]),
                name=a["AreaName_lang"],
                map_id=map_id,
                map_name=maps.get(map_id, ""),
                parent_id=_int_or_none(a["ParentAreaID"]),
                level_min=level if level > 0 else None,
                level_max=None,
            )
        )
    return zones
```

```python
# data/pipeline/normalize/dungeons.py
from pipeline.models import Dungeon


def normalize_dungeons(journal_rows: list[dict[str, str]], map_rows: list[dict[str, str]]) -> list[Dungeon]:
    known_maps = {int(m["ID"]) for m in map_rows}
    out: list[Dungeon] = []
    for j in journal_rows:
        map_id = int(j["MapID"])
        # keep entries even if the Map fixture lacks them; Map is used only for validation logging in Phase 0
        _ = map_id in known_maps
        out.append(
            Dungeon(
                id=int(j["ID"]),
                name=j["Name_lang"],
                map_id=map_id,
                description=j.get("Description_lang", ""),
                order=int(j.get("OrderIndex", "0") or 0),
            )
        )
    return out
```

```python
# data/pipeline/normalize/items.py
from pipeline.models import Item


def normalize_items(sparse_rows: list[dict[str, str]], item_rows: list[dict[str, str]]) -> list[Item]:
    classes = {int(r["ID"]): (int(r["ClassID"]), int(r["SubclassID"])) for r in item_rows}
    out: list[Item] = []
    for s in sparse_rows:
        item_id = int(s["ID"])
        if item_id not in classes:
            continue
        class_id, subclass_id = classes[item_id]
        out.append(
            Item(
                id=item_id,
                name=s["Display_lang"],
                quality=int(s["OverallQualityID"]),
                item_level=int(s["ItemLevel"]),
                required_level=int(s["RequiredLevel"]),
                class_id=class_id,
                subclass_id=subclass_id,
                inventory_type=int(s["InventoryType"]),
            )
        )
    return out
```

```python
# data/pipeline/normalize/classes.py
from pipeline.models import PlayableClass, PlayableRace

CLASS_COLORS = {
    "warrior": "#c69b6d", "paladin": "#f48cba", "hunter": "#aad372", "rogue": "#fff468",
    "priest": "#ffffff", "shaman": "#0070dd", "mage": "#3fc7eb", "warlock": "#8788ee", "druid": "#ff7c0a",
}


def _slug(name: str) -> str:
    return name.lower().replace(" ", "-").replace("'", "")


def normalize_classes(rows: list[dict[str, str]]) -> list[PlayableClass]:
    out: list[PlayableClass] = []
    for r in rows:
        slug = _slug(r["Name_lang"])
        out.append(PlayableClass(id=int(r["ID"]), name=r["Name_lang"], slug=slug, color=CLASS_COLORS.get(slug, "#ffffff")))
    return out


def normalize_races(rows: list[dict[str, str]]) -> list[PlayableRace]:
    faction_by_flag = {"0": "alliance", "1": "horde", "2": "neutral"}
    return [
        PlayableRace(id=int(r["ID"]), name=r["Name_lang"], slug=_slug(r["Name_lang"]), faction=faction_by_flag.get(r["Alliance"].strip(), "neutral"))
        for r in rows
    ]
```

```python
# data/pipeline/manifest.py
import hashlib
import json
from pathlib import Path

MANIFEST = "manifest.json"


def _sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write_manifest(build_dir: Path, build: str, product: str, fetched_at: str) -> dict:
    files = {p.name: _sha256(p) for p in sorted(build_dir.glob("*.json")) if p.name != MANIFEST}
    m = {"build": build, "product": product, "fetched_at": fetched_at, "files": files}
    (build_dir / MANIFEST).write_text(json.dumps(m, indent=2) + "\n", encoding="utf-8")
    return m


def read_manifest(build_dir: Path) -> dict:
    return json.loads((build_dir / MANIFEST).read_text(encoding="utf-8"))


def verify(build_dir: Path) -> list[str]:
    m = read_manifest(build_dir)
    return [name for name, digest in m["files"].items() if not (build_dir / name).exists() or _sha256(build_dir / name) != digest]
```

```python
# data/pipeline/wago.py
import json
from datetime import UTC, datetime
from pathlib import Path

import httpx

BASE_URL = "https://wago.tools"
TABLES = ["Map", "AreaTable", "JournalInstance", "ItemSparse", "Item", "SpellName", "ChrClasses", "ChrRaces", "Talent", "TalentTab"]


def latest_build(product: str, client: httpx.Client) -> str:
    r = client.get("/api/builds")
    r.raise_for_status()
    builds = r.json().get(product)
    if not builds:
        raise SystemExit(f"wago.tools lists no builds for product {product!r}; known products: {sorted(r.json())}")
    return builds[0]["version"]


def download_table(table: str, build: str, dest: Path, client: httpx.Client) -> Path:
    r = client.get(f"/db2/{table}/csv", params={"build": build}, timeout=120)
    r.raise_for_status()
    if not r.text.startswith("ID"):
        raise SystemExit(f"{table} for build {build} did not return a CSV with an ID column; got: {r.text[:80]!r}")
    dest.mkdir(parents=True, exist_ok=True)
    path = dest / f"{table}.csv"
    path.write_text(r.text, encoding="utf-8")
    return path


def fetch_build(product: str, build: str | None, root: Path = Path("builds"), client: httpx.Client | None = None) -> Path:
    own = client is None
    client = client or httpx.Client(base_url=BASE_URL, headers={"User-Agent": "foreversixty-pipeline/0.1 (+https://foreversixty.gg)"})
    try:
        build = build or latest_build(product, client)
        raw = root / build / "raw"
        for table in TABLES:
            download_table(table, build, raw, client)
        (raw / "_meta.json").write_text(
            json.dumps({"product": product, "build": build, "fetched_at": datetime.now(UTC).isoformat().replace("+00:00", "Z")}, indent=2) + "\n"
        )
        return root / build
    finally:
        if own:
            client.close()
```

```python
# data/pipeline/diff.py
import json
from pathlib import Path

ENTITIES = ["zones", "dungeons", "items", "spells", "classes", "races", "talents"]


def diff_entities(before: list[dict], after: list[dict]) -> dict:
    b = {r["id"]: r for r in before}
    a = {r["id"]: r for r in after}
    added = [a[i] for i in sorted(a.keys() - b.keys())]
    removed = [b[i] for i in sorted(b.keys() - a.keys())]
    changed = []
    for i in sorted(a.keys() & b.keys()):
        fields = sorted(k for k in set(a[i]) | set(b[i]) if a[i].get(k) != b[i].get(k))
        if fields:
            changed.append({"id": i, "before": b[i], "after": a[i], "fields": fields})
    return {"added": added, "removed": removed, "changed": changed}


def diff_builds(from_build: str, to_build: str, root: Path = Path("builds"), out: Path = Path("diffs")) -> Path:
    result = {"from": from_build, "to": to_build, "entities": {}}
    for name in ENTITIES:
        fb, tb = root / from_build / f"{name}.json", root / to_build / f"{name}.json"
        if not fb.exists() or not tb.exists():
            continue
        d = diff_entities(json.loads(fb.read_text()), json.loads(tb.read_text()))
        result["entities"][name] = d
        print(f"{name}: +{len(d['added'])} -{len(d['removed'])} ~{len(d['changed'])}")
    out.mkdir(parents=True, exist_ok=True)
    path = out / f"{from_build}__{to_build}.json"
    path.write_text(json.dumps(result, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    return path
```

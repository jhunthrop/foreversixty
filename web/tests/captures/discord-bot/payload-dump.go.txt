type dumped struct {
	ID      string          `json:"id"`

// capPort is the real guilds service with the reads a test database cannot back scripted.
type capPort struct {
	*guilds.Service
	clock       *time.Time
	officerID   int64
	signUpErr   error
	readinessBy func(viewer int64) (guilds.ReadinessAnswer, error)
	report      *guilds.ReportAnswer
	attendance  func(viewer int64, name string) ([]guilds.AttendanceRecord, error)
	facts       map[string]guilds.ReadinessFacts
	done        *notify.RaidPost
}

func (p *capPort) SignUp(ctx context.Context, cmd guilds.SignupCommand) (guilds.SignupResult, error) {
	if p.signUpErr != nil {
		return guilds.SignupResult{}, p.signUpErr
	}
	return p.Service.SignUp(ctx, cmd)
}

func (p *capPort) NextRaidReadiness(_ context.Context, _, viewer int64, _ time.Time) (guilds.ReadinessAnswer, error) {
	return p.readinessBy(viewer)
}

func (p *capPort) ReportFor(ctx context.Context, guildID, viewerID int64, _ *int64) (guilds.ReportAnswer, bool, error) {
	_, member, err := p.Accounts.GuildRank(ctx, guildID, viewerID)
	if err != nil {
		return guilds.ReportAnswer{}, false, err
	}
	if p.report == nil || (!member && !p.report.Public) {
		return guilds.ReportAnswer{}, false, nil
	}
	ans := *p.report
	ans.Member = member
	ans.Officer = member && viewerID == p.officerID
	return ans, true, nil
}

func (p *capPort) AttendanceFor(_ context.Context, _, viewer int64, name string) ([]guilds.AttendanceRecord, error) {
	return p.attendance(viewer, name)
}

func (p *capPort) CharacterReadinessFacts(_ context.Context, _ int64, key string) (guilds.ReadinessFacts, error) {
	return p.facts[key], nil
}

// RaidPostFacts is the real read; a done raid adds what its matched log would (Came, Kills).
func (p *capPort) RaidPostFacts(ctx context.Context, raidID int64) (guilds.RaidPostFacts, bool, error) {
	f, ok, err := p.Service.RaidPostFacts(ctx, raidID)
	if err == nil && ok && p.done != nil && f.Status == guilds.RaidDone {
		f.Post.Came, f.Post.CameOf, f.Post.Kills = p.done.Came, p.done.CameOf, p.done.Kills
	}
	return f, ok, err
}

type capCards struct{ cards map[string]rankings.Character }

func (c *capCards) CharacterCard(_ context.Context, region, ruleset, name string) (rankings.Character, bool, error) {
	card, ok := c.cards[strings.ToLower(name)]
	if !ok || card.Character.Region != region || card.Character.Ruleset != ruleset {
		return rankings.Character{}, false, nil
	}
	return card, true, nil
}

type capWorld struct {
	t      *testing.T
	st     *Store
	gs     *guilds.Service

	svc    *Service
	rest   *FakeREST
	clock  time.Time

	zul    int64
	raidID int64
	start  time.Time
	out    []dumped
	row    string
}

func (w *capWorld) exec(q string, args ...any) {
	w.t.Helper()
	if _, err := w.st.Pool.Exec(context.Background(), q, args...); err != nil {
		w.t.Fatal(err)
	}
}

func (w *capWorld) add(id, title, surface string, m Message) {
	w.t.Helper()
	raw, err := json.Marshal(m)
	if err != nil {
		w.t.Fatal(err)
	}
	w.out = append(w.out, dumped{ID: id, Row: w.row, Title: title, Surface: surface, Clock: w.clock, Message: raw})
}

func (w *capWorld) reply(id, title string, rep Reply, err error) Reply {
	w.t.Helper()
	if err != nil {
		w.t.Fatal(err)
	}
	w.add(id, title, "ephemeral", rep.Message)
	return rep
}

func (w *capWorld) in(kind int, discordUser string, data any) Interaction {
	raw, _ := json.Marshal(data)
	return Interaction{ID: "i1", Token: "tok", GuildID: capServer, DiscordUserID: discordUser, Type: kind, Data: raw}
}

func (w *capWorld) press(user, state string) (Reply, error) {
	return w.svc.press(context.Background(), w.in(typeComponent, user, map[string]any{"custom_id": raidCustomID(w.raidID, state)}))
}

func (w *capWorld) pick(state, key string) (Reply, error) {
	return w.svc.pick(context.Background(), w.in(typeComponent, dZul,
		map[string]any{"custom_id": pickCustomID(w.raidID, state), "values": []string{key}}))
}

func (w *capWorld) command(user, name string, options map[string]string) (Reply, error) {
	opts := []map[string]any{}
	for k, v := range options {
		opts = append(opts, map[string]any{"name": k, "value": v})
	}
	return w.svc.commands[name](context.Background(), w.in(typeCommand, user, map[string]any{"name": name, "options": opts}))
}

func (w *capWorld) user(email string) int64 {
	w.t.Helper()
	var id int64
	if err := w.st.Pool.QueryRow(context.Background(), `insert into users (email) values ($1) returning id`, email).Scan(&id); err != nil {
		w.t.Fatal(err)
	}
	return id
}

// char puts a character of userID on the roster, its spec from a logged fight.
func (w *capWorld) char(userID int64, key, name, class, spec, rank string, verified bool) {
	w.t.Helper()
	w.exec(`insert into characters (key, region, ruleset, name, class, user_id) values ($1, 'us', 'normal', $2, $3, $4)`,
		key, name, strings.ToLower(class), userID)
	w.exec(`insert into guild_characters (guild_id, character_key, user_id, rank, verified_at)
		values ($1, $2, $3, $4, case when $5 then now() else null end)`, w.gid, key, userID, rank, verified)
	w.exec(`insert into fight_metrics (report_id, fight_index, player_key, class, spec, ilvl, fought_at)
		values ('capfight', (select count(*) from fight_metrics where report_id = 'capfight'), $1, $2, $3, 60, now())`, key, class, spec)
	ctx := context.Background()
	tx, err := w.st.Pool.Begin(ctx)
	if err != nil {
		w.t.Fatal(err)
	}
	defer func() { _ = tx.Rollback(ctx) }()
	if err := guilds.RecomputeMembership(ctx, tx, w.gid, &userID); err != nil {
		w.t.Fatal(err)
	}
	if err := tx.Commit(ctx); err != nil {
		w.t.Fatal(err)
	}
}

// post renders the raid's post as the drainer sends it, from a fresh post row, at the clock.
func (w *capWorld) post(id, title string, moved *time.Time) {
	w.t.Helper()
	ctx := context.Background()
	w.exec(`delete from guild_raid_discord_posts where raid_id = $1`, w.raidID)
	if ok, err := w.st.EnqueuePost(ctx, w.raidID, w.gid); err != nil || !ok {
		w.t.Fatalf("enqueue %v %v", ok, err)
	}
	if moved != nil {
		if err := w.st.SetMovedFrom(ctx, w.raidID, *moved); err != nil {
			w.t.Fatal(err)
		}
	}
	before := len(w.rest.Calls())
	d := w.svc.NewDrainer(w.port)
	d.Now = func() time.Time { return w.clock }
	if err := d.DrainOnce(ctx); err != nil {
		w.t.Fatal(err)
	}
	calls := w.rest.Calls()[before:]
	for _, c := range calls {
		if c.Method == "CreateMessage" {
			w.add(id, title, "channel", c.Message)
			return
		}
	}
	w.t.Fatalf("%s: no post sent: %+v", id, calls)
}

func at(layout string) time.Time {
	t, err := time.Parse(time.RFC3339, layout)
	if err != nil {
		panic(err)
	}
	return t
}

// seedSignups is the rest of the raid before Zulmara: 30 In (3 tanks, 8 healers, 19 DPS),
// 2 Late, 3 Bench and 4 Out, so Zulmara's In makes 33 of 40 · 3 tanks · 8 healers · 20 DPS.
func seedSignups(w *capWorld, filler int64) {
	type s struct{ name, class, spec, role, state string }
	rows := []s{
		{"Kraggor", "warrior", "Protection", "tank", "in"}, {"Bramblehide", "druid", "Feral", "tank", "in"},
		{"Ironvale", "warrior", "Protection", "tank", "in"},
	}
	healers := []string{"Lightwell", "Aurelith", "Mossgrove", "Dawnmere", "Sablewind", "Halcyra", "Embertide", "Vaelora"}
	healerClass := []string{"priest", "paladin", "druid", "priest", "shaman", "paladin", "priest", "druid"}
	for i, n := range healers {
		rows = append(rows, s{n, healerClass[i], "Holy", "healer", "in"})
	}
	dps := []string{"Pyrewisp", "Windtalon", "Vexlash", "Grimhollow", "Ashenra", "Thornquill", "Nightbrand", "Corvane",
		"Frostwhisk", "Duskmantle", "Ravenmoor", "Stormvale", "Brightshot", "Kelthar", "Orivane", "Sundermark",
		"Wyrmbane", "Cindrel", "Talvorn"}
	for _, n := range dps {
		rows = append(rows, s{n, "mage", "Fire", "dps", "in"})
	}
	rows = append(rows, s{"Glimmerfen", "warlock", "Destruction", "dps", "late"}, s{"Hollowmere", "rogue", "Combat", "dps", "late"})
	for _, n := range []string{"Quillfeather", "Marrowind", "Selwyn"} {
		rows = append(rows, s{n, "hunter", "Marksmanship", "dps", "bench"})
	}
	for _, n := range []string{"Fennrick", "Ysolde", "Garrowe", "Lunetta"} {
		rows = append(rows, s{n, "warlock", "Affliction", "dps", "out"})
	}
	for _, r := range rows {
		w.exec(`insert into guild_raid_signups (raid_id, character_key, name, class, spec, role, state, signed_by, acted_by)
			values ($1, $2, $3, $4, $5, $6, $7, $8, $8)`, w.raidID, "us/normal/"+strings.ToLower(r.name), r.name, r.class, r.spec, r.role, r.state, filler)
	}
}

// zulmaraCapCard is Zulmara's public card: a level 60 Night Elf Beast Mastery Hunter wearing
// the bare level-60 Alliance band's picks, except three slots where she wears the band's own
// first alternative. Names, sources and item levels are the build's; the gaps only order the
// three (the DPS each shows is the BiS file's measured delta for that very pair).
func zulmaraCapCard(t *testing.T, w *capWorld, dataDir, build string) rankings.Character {
	band, found, err := bis.LoadPresetBand(dataDir, w.gs.BisBuild(), capSpec, 60, "alliance", bis.BarePreset)
	if err != nil || !found {
		t.Fatalf("band %v %v", found, err)
	}
	wornAlt := map[string]bool{"legs": true, "finger1": true, "chest": true}
	var gear []rankings.GearSlot
	for _, sl := range band.Slots {
		if sl.ItemID <= 0 {
			continue
		}
		pick := &rankings.GearPick{ID: sl.ItemID, Name: sl.ItemName, Source: sl.Source}
		wornID, gap := sl.ItemID, 0.0
		if wornAlt[sl.Slot] && len(sl.Alternatives) > 0 {
			wornID, gap = sl.Alternatives[0].ItemID, -sl.Alternatives[0].DpsDelta
		}
		it, ok := w.gs.BuildItem(build, "hunter", wornID)
		if !ok {
			t.Fatalf("item %d not in the build", wornID)
		}
		name, ilvl := it.Name, it.ItemLevel
		gear = append(gear, rankings.GearSlot{Slot: sl.Slot, Worn: &rankings.GearWorn{ID: wornID, Name: &name, ItemLevel: &ilvl},
			Pick: pick, Gap: ref(gap)})
	}
	return rankings.Character{
		Character: rankings.CharacterRef{Key: keyZul, Region: "us", Ruleset: "normal", Name: "Zulmara", Class: "hunter",
			Level: ref(60), Race: ref("Night Elf"), Guild: &rankings.CharacterGuild{Name: "Olympus XXVII"}},
		GearSpec: ref(capSpec), Gear: gear,
	}
}

// shareFrames presses Share in channel on answer: the public follow-up, then the ephemeral
// with its Shared button.
func shareFrames(w *capWorld, answer Reply, command, presser string) {
	w.t.Helper()
	var id string
	for _, row := range answer.Message.Components {
		for _, c := range row {
			if strings.HasPrefix(c.CustomID, shareIDPrefix) {
				id = c.CustomID
			}
		}
	}
	if id == "" {
		w.t.Fatalf("%s: no share button", command)
	}
	in := w.in(typeComponent, presser, map[string]any{"custom_id": id})
	in.Token = "tok-share-" + command
	rep, err := w.svc.share(context.Background(), in)
	if err != nil {
		w.t.Fatal(err)
	}
	if rep.After != nil {
		rep.After(context.Background())
	}
	for _, c := range w.rest.Calls() {
		if c.Method == "FollowUp" && c.Token == in.Token {
			w.add(command+"-shared", "/"+command+" shared in the channel", "channel", c.Message)
		}
	}
	w.add(command+"-after-share", "/"+command+" ephemeral after sharing", "ephemeral", rep.Message)
}

	w.reply("press-moved", "Press: In on the other character, the R3 move", rep, err)
	// Back to the example: Zulmara In, Duskstrider not signed.
	w.exec(`delete from guild_raid_signups where raid_id = $1 and character_key = $2`, w.raidID, keyDuskC)
	w.exec(`update guild_raid_signups set state = 'in' where raid_id = $1 and character_key = $2`, w.raidID, keyZul)

	w.row = "D4"
	rep, err = w.command(dZul, "signup", map[string]string{"status": "late", "character": keyZul})
	w.reply("signup", "/signup status:Late character:Zulmara", rep, err)
	w.exec(`update guild_raid_signups set state = 'in' where raid_id = $1 and character_key = $2`, w.raidID, keyZul)

	w.row = "D3"
	w.port.signUpErr = errors.New("capture: the save failed")
	rep, err = w.pick("in", keyZul)
	w.reply("press-failure", "Press: the save failed", rep, err)
	w.port.signUpErr = nil
	for i := 0; i < 60; i++ {
		rep, err = w.pick("in", keyZul)
		if err != nil {
			w.t.Fatal(err)
		}
		if rep.Message.Content == pressCopy.RateLimited {
			w.reply("press-rate-limit", "Press: rate limit", rep, nil)
			break
		}
	}
	w.gs.SignupLimiter = httpx.NewKeyLimiter(1000, time.Minute)
}

func officerFrames(w *capWorld, officer int64, build string) {
	ctx := context.Background()
	raid, found, err := w.gs.GetRaid(ctx, w.gid, w.raidID)
	if err != nil || !found {
		w.t.Fatal(err)
	}
	// /report at the done clock.
	w.row = "D7"
	w.port.report = capReportAnswer(w, raid, build)
	rep, err := w.command(dZul, "report", nil)
	w.reply("report-member", "/report, a member", rep, err)
	rep, err = w.command(dOfficer, "report", nil)
	w.reply("report-officer", "/report, an officer", rep, err)
	shareFrames(w, rep, "report", dOfficer)

	// /attendance and /readiness before the raid, Thursday 4:00 pm CST.
	w.clock = at("2026-12-10T22:00:00Z")
	w.exec(`update guild_raids set status = 'planned' where id = $1`, w.raidID)
	raid.Status = guilds.RaidPlanned
	w.row = "D8"
	w.port.attendance = capAttendance(w, officer)
	rep, err = w.command(dZul, "attendance", nil)
	w.reply("attendance-own", "/attendance, own (two characters)", rep, err)
	rep, err = w.command(dOfficer, "attendance", map[string]string{"name": "Zulmara"})
	w.reply("attendance-officer", "/attendance name:Zulmara, an officer", rep, err)
	rep, err = w.command(dZul, "attendance", map[string]string{"name": "Kraggor"})
	w.reply("attendance-refusal", "/attendance name:Kraggor, a member", rep, err)

	w.row = "D9"
	board := guilds.CaptureReadinessAnswer(raid, capReadinessView(raid))
	mode := "officer"
	w.port.readinessBy = func(viewer int64) (guilds.ReadinessAnswer, error) {
		switch {
		case viewer != officer:
			return guilds.ReadinessAnswer{}, guilds.ErrNotOfficer
		case mode == "lapsed":
			return guilds.ReadinessAnswer{}, guilds.ErrPlanRequired
		case mode == "guild-tier":
			return guilds.ReadinessAnswer{}, guilds.ErrRaidGuildRequired
		case mode == "nobody":
			return guilds.ReadinessAnswer{Raid: &raid, Rows: []string{}}, nil
		}
		return board, nil
	}
	rep, err = w.command(dOfficer, "readiness", nil)
	w.reply("readiness-officer", "/readiness, an officer", rep, err)
	rep, err = w.command(dZul, "readiness", nil)
	w.reply("readiness-member", "/readiness, a member", rep, err)
	rep, err = w.command(dStranger, "readiness", nil)
	w.reply("readiness-unlinked", "/readiness, not linked", rep, err)
	mode = "lapsed"
	rep, err = w.command(dOfficer, "readiness", nil)
	w.reply("readiness-lapsed", "/readiness, plan lapsed", rep, err)
	mode = "guild-tier"
	rep, err = w.command(dOfficer, "readiness", nil)
	w.reply("readiness-guild-tier", "/readiness, Guild tier", rep, err)
	mode = "nobody"
	rep, err = w.command(dOfficer, "readiness", nil)
	w.reply("readiness-nobody", "/readiness, nobody signed In or Late", rep, err)
}

// capReportAnswer is the done raid's matched log: Onyxia killed after two wipes, and four of
// her real drops (the build's loot table), awarded.
func capReportAnswer(w *capWorld, raid guilds.Raid, build string) *guilds.ReportAnswer {
	zone := capZone
	kill := int64(4*60_000 + 12_000)
	award := func(id int, name, class string) guilds.ReportLootAward {
		it, ok := w.gs.BuildItem(build, class, id)
		if !ok {
			w.t.Fatalf("loot item %d not in the %s table", id, class)
		}
		return guilds.ReportLootAward{ItemID: id, ItemName: it.Name, Quality: it.Quality, Icon: it.Icon, Name: name, Class: class}
	}
	flask, noFlask := true, false
	return &guilds.ReportAnswer{
		Raid: guilds.RaidView{ID: raid.ID, Title: raid.Title, Status: guilds.RaidDone, GuildTimezone: &zone, StartsAt: raid.StartsAt,
			Match: &guilds.MatchView{ReportID: capReport},
			Night: &guilds.NightView{Signed: 33, Came: 30, Kills: 1, Wipes: 2, Deaths: 11, DurationMS: 52 * 60_000}},
		Report: &guilds.ReportView{
			KillOrder: []guilds.KillOrderView{{Name: "Onyxia", Killed: true, KillMS: &kill, WipesBefore: 2}},
			Deaths: []guilds.ReportDeathView{{Name: "Pyrewisp", OnWipes: 2, OnKills: 1}, {Name: "Windtalon", OnWipes: 2},
				{Name: "Vexlash", OnWipes: 1, OnKills: 1}, {Name: "Corvane", OnWipes: 1}},
			Consumables: guilds.ReportConsumablesView{LogHasBuffs: true, Rows: []guilds.ConsumableRow{
				{Name: "Vexlash", Before: "short", Flask: &noFlask}, {Name: "Corvane", Before: "short", Flask: &noFlask},
				{Name: "Talvorn", Before: "short", Flask: &flask}, {Name: "Pyrewisp", Before: "stocked", Flask: &flask}}},
			Loot: []guilds.ReportLootBoss{{Boss: "Onyxia", Awards: []guilds.ReportLootAward{
				award(16939, "Zulmara", "hunter"), award(17078, "Pyrewisp", "mage"),
				award(18813, "Kraggor", "warrior"), award(18205, "Hollowmere", "rogue")}}},
		},
		Public: true,
	}
}

// capAttendance is the roster's attendance: Zulmara 7 of the last 8, 26 of 30; Duskstrider,
// a newer alt, 2 of 2. A member naming someone else is refused, as AttendanceFor refuses.
func capAttendance(w *capWorld, officer int64) func(int64, string) ([]guilds.AttendanceRecord, error) {
	day := func(d int) time.Time { return w.start.AddDate(0, 0, -7*d) }
	zul := guilds.AttendanceRecord{Name: "Zulmara", Class: "hunter",
		Chip: guilds.CaptureAttendanceChip(guilds.RosterRow{Attendance: guilds.Attendance{Present: 7, Nights: 8},
			Attendance30: &guilds.Attendance{Present: 26, Nights: 30}}),
		Last: []guilds.AttendanceLine{{StartsAt: day(1), Outcome: "present"}, {StartsAt: day(2), Outcome: "late", Excused: true},
			{StartsAt: day(3), Outcome: "present"}, {StartsAt: day(4), Outcome: "no_show"}, {StartsAt: day(5), Outcome: "present"}}}
	dusk := guilds.AttendanceRecord{Name: "Duskstrider", Class: "rogue",
		Chip: guilds.CaptureAttendanceChip(guilds.RosterRow{Attendance: guilds.Attendance{Present: 2, Nights: 2}}),
		Last: []guilds.AttendanceLine{{StartsAt: day(6), Outcome: "present"}, {StartsAt: day(7), Outcome: "present"}}}
	return func(viewer int64, name string) ([]guilds.AttendanceRecord, error) {
		switch {
		case name == "":
			return []guilds.AttendanceRecord{zul, dusk}, nil
		case viewer == officer && strings.EqualFold(name, "Zulmara"):
			return []guilds.AttendanceRecord{zul}, nil
		case viewer != officer:
			return nil, guilds.ErrOfficerOnly
		}
		return nil, guilds.ErrNoSuchMember
	}
}

// capReadinessView is the board for the 33 signed In or Late, four of them with something to fix.
func capReadinessView(raid guilds.Raid) guilds.ReadinessView {
	stale := false
	rows := []guilds.ReadinessRow{
		{Name: "Pyrewisp", Consumables: guilds.ConsumablesView{State: "short"}, SyncedBeforeCutoff: &stale,
			LoggedAt: raid.StartsAt.Add(-9 * 24 * time.Hour), Failing: 2},
		{Name: "Windtalon", GearGap: &guilds.GearGapView{Upgrades: 2, GainDps: 14.6}, Enchants: guilds.EnchantsView{Checked: true,
			MissingSlots: []string{"chest", "feet"}}, Failing: 2},
		{Name: "Kraggor", TalentPointsUnspent: 3, TalentPointsRead: true, Failing: 1},
		{Name: "Vexlash", NoExport: true},
	}
	for i := len(rows); i < 33; i++ {
		rows = append(rows, guilds.ReadinessRow{Name: fmt.Sprintf("Ready%d", i)})
	}
	return guilds.ReadinessView{Rows: rows}
}
